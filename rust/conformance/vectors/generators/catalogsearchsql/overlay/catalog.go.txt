// VecFuzzyRerank reranks items titled titles[i] (content ID "i") with aliases[i], and
// returns the admitted content IDs in order plus required.
func VecFuzzyRerank(searchText string, titles []string, aliases [][]string, exactHits bool) ([]string, int) {
	items := make([]*models.MediaItem, len(titles))
	byID := map[string][]string{}
	for i, t := range titles {
		id := strconv.Itoa(i)
		items[i] = &models.MediaItem{ContentID: id, Title: t}
		if i < len(aliases) && aliases[i] != nil {
			byID[id] = aliases[i]
		}
	}
	ranks, required, _ := rankFuzzyItems(context.Background(), searchText, items, byID)
	ranks = admitFuzzyRanks(ranks, required, exactHits)
	out := []string{}
	for _, r := range ranks {
		out = append(out, r.item.ContentID)
	}
	return out, required
}
